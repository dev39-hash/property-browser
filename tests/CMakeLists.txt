# qpb_add_test(<name> SOURCES <files...> LIBRARIES <targets...>)
#
# Builds a Qt Test executable and registers it with CTest. Widget tests run on
# the offscreen platform so they need no display.
function(qpb_add_test name)
    cmake_parse_arguments(PARSE_ARGV 1 arg "" "" "SOURCES;LIBRARIES")
    add_executable(${name} ${arg_SOURCES})
    target_link_libraries(${name} PRIVATE Qt${QT_VERSION_MAJOR}::Test ${arg_LIBRARIES})
    set_target_properties(${name} PROPERTIES AUTOMOC ON)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        # Qt 6.5's QTEST_*_MAIN macros trip this pedantic check.
        target_compile_options(${name} PRIVATE -Wno-gnu-zero-variadic-macro-arguments)
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set(environment "QT_QPA_PLATFORM=offscreen")
    if(WIN32 AND QT_VERSION_MAJOR EQUAL 5)
        # Qt 5's offscreen platform finds no fonts on Windows and draws no text.
        list(APPEND environment "QT_QPA_FONTDIR=$ENV{WINDIR}/Fonts")
    endif()
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "${environment}")
endfunction()

# qpb/translations/*.ts are complete and up to date with the sources. Needs Qt
# Linguist's lupdate, which building qpb itself never does: the .qm files are
# generated by the maintainers and committed, because a consumer may not have
# LinguistTools installed.
find_package(Qt${QT_VERSION_MAJOR} QUIET COMPONENTS LinguistTools)
find_package(Python3 COMPONENTS Interpreter)
if(TARGET Qt${QT_VERSION_MAJOR}::lupdate AND Python3_Interpreter_FOUND)
    add_test(NAME translations_up_to_date
        COMMAND Python3::Interpreter "${PROJECT_SOURCE_DIR}/tools/update_translations.py"
            --qt-bin "$<TARGET_FILE_DIR:Qt${QT_VERSION_MAJOR}::lupdate>" --check
    )
else()
    message(STATUS "lupdate or Python 3 not found: the translations_up_to_date test is not available")
endif()
